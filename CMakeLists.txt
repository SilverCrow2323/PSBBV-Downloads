cmake_minimum_required(VERSION 3.16)
project(psbbv_downloads_core C)
set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
add_library(psbbv_transfer_job src/transfer_job.c)
target_include_directories(psbbv_transfer_job PUBLIC include)
target_compile_options(psbbv_transfer_job PRIVATE -Wall -Wextra -Wpedantic -Werror)
enable_testing()
add_executable(test_transfer_job tests/test_transfer_job.c)
target_link_libraries(test_transfer_job PRIVATE psbbv_transfer_job)
target_compile_options(test_transfer_job PRIVATE -Wall -Wextra -Wpedantic -Werror)
add_test(NAME transfer_job COMMAND test_transfer_job)
